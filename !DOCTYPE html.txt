      }
    ];

    const mockVocabs = [
      {
        id: 'voc-301',
        phrase: 'Scope Creep',
        phonetic: '/skoʊp kriːp/',
        meaning: 'Hiện tượng dự án bị phình to phạm vi so với thỏa thuận ban đầu',
        context: 'To avoid scope creep, we need to formally document these additional requests in a Change Order.',
        isSaved: true
      },
      {
        id: 'voc-302',
        phrase: 'Push back constructively',
        phonetic: '/pʊʃ bæk kənˈstrʌktɪvli/',
        meaning: 'Phản hồi từ chối một cách khéo léo, mang tính xây dựng',
        context: 'The lead consultant had to push back constructively when the client requested a 50% shorter deadline.',
        isSaved: false
      },
      {
        id: 'voc-303',
        phrase: 'Align on expectations',
        phonetic: '/əˈlaɪn ɒn ˌɛkspekˈteɪʃənz/',
        meaning: 'Thống nhất và đồng thuận về kỳ vọng đôi bên',
        context: 'Before proceeding with the audit, let us align on the expected deliverables.',
        isSaved: false
      },

        id: 'voc-304',
        phrase: 'Bottom-line upfront (BLUF)',
        phonetic: '/ˈbɑː.t̬əm laɪn ʌpˈfrʌnt/',
        meaning: 'Đưa kết luận/khuyến nghị quan trọng nhất lên đầu câu chuyện',
        context: 'In our executive briefing, always use the BLUF principle to respect the CEO\'s limited time.',
        isSaved: true
      }
    ];

    let currentVideoIndex = 0;

    // Render danh sách video bài học
    function renderVideoList() {
      const container = document.getElementById('video-list');
      container.innerHTML = '';
      
      let completedCount = 0;

      mockVideos.forEach((vid, index) => {
        if (vid.isCompleted) completedCount++;
        const isCurrent = index === currentVideoIndex;

        const item = document.createElement('div');
        item.className = `p-3 rounded-xl cursor-pointer border transition flex items-start gap-3 ${
          isCurrent 
            ? 'bg-blue-50/70 border-blue-200' 
            : 'bg-white border-slate-100 hover:border-slate-200 hover:bg-slate-50'
        }`;
        
        item.onclick = () => selectVideo(index);

        item.innerHTML = `
          <div class="mt-0.5">
            <i class="ph-bold ${vid.isCompleted ? 'ph-check-circle text-emerald-600' : 'ph-circle text-slate-300'} text-base"></i>
          </div>
          <div class="flex-1 min-w-0">
            <p class="text-xs font-semibold ${isCurrent ? 'text-blue-900 font-bold' : 'text-slate-800'} line-clamp-1">${vid.title}</p>
            <p class="text-[11px] text-slate-400 mt-0.5">${vid.duration} • ${vid.channel}</p>
          </div>
        `;
        container.appendChild(item);
      });

      document.getElementById('completed-count').innerText = `${completedCount}/${mockVideos.length} Đã xong`;
    }

    // Chọn video xem
    function selectVideo(index) {
      currentVideoIndex = index;
      const vid = mockVideos[index];

      document.getElementById('video-player').src = `https://www.youtube.com/embed/${vid.youtubeId}?autoplay=1`;
      document.getElementById('video-title').innerText = vid.title;
      document.getElementById('video-meta').innerText = `Kênh: ${vid.channel} • Thời lượng: ${vid.duration}`;
      document.getElementById('video-takeaway').innerText = vid.takeaway;

      updateCompleteButtonState();
      renderVideoList();
    }

    // Cập nhật trạng thái nút hoàn thành
    function updateCompleteButtonState() {
      const btn = document.getElementById('btn-complete-lesson');
      const isDone = mockVideos[currentVideoIndex].isCompleted;

      if (isDone) {
        btn.className = "inline-flex items-center justify-center gap-2 px-4 py-2 rounded-xl text-sm font-semibold transition-all border border-emerald-300 bg-emerald-50 text-emerald-700";
        btn.innerHTML = `<i class="ph-bold ph-check text-emerald-600"></i> Đã hoàn thành`;
      } else {
        btn.className = "inline-flex items-center justify-center gap-2 px-4 py-2 rounded-xl text-sm font-semibold transition-all shadow-sm bg-blue-600 hover:bg-blue-700 text-white";
        btn.innerHTML = `<i class="ph-bold ph-check-circle"></i> Đánh dấu hoàn thành`;
      }
    }

    // Đánh dấu hoàn thành bài học hiện tại
    function toggleCompleteCurrentVideo() {
      mockVideos[currentVideoIndex].isCompleted = !mockVideos[currentVideoIndex].isCompleted;
      updateCompleteButtonState();
      renderVideoList();
    }

    // Render danh sách từ vựng chuyên ngành
    function renderVocabList() {
      const container = document.getElementById('vocab-container');
      container.innerHTML = '';
      let savedCount = 0;

      mockVocabs.forEach((v, idx) => {
        if (v.isSaved) savedCount++;

        const card = document.createElement('div');
        card.className = "p-3 rounded-xl border border-slate-100 bg-slate-50/50 hover:bg-slate-50 transition space-y-1.5";
        card.innerHTML = `
          <div class="flex items-center justify-between">
            <h4 class="text-xs font-bold text-slate-900">${v.phrase}</h4>
            <button onclick="toggleSaveVocab(${idx})" class="text-slate-400 hover:text-blue-600 transition">
              <i class="ph-bold ${v.isSaved ? 'ph-bookmark-simple text-blue-600' : 'ph-bookmark-simple'} text-base"></i>
            </button>
          </div>
          <p class="text-[11px] text-slate-500 font-mono">${v.phonetic}</p>
          <p class="text-xs text-slate-700 font-medium">${v.meaning}</p>
          <p class="text-[11px] text-slate-500 italic bg-white p-2 rounded-lg border border-slate-100">"${v.context}"</p>
        `;
        container.appendChild(card);
      });

      document.getElementById('saved-vocab-count').innerText = savedCount;
    }

    // Lưu / bỏ lưu từ vựng
    function toggleSaveVocab(index) {
      mockVocabs[index].isSaved = !mockVocabs[index].isSaved;
      renderVocabList();
    }

    // Kiểm tra câu trả lời quiz
    function checkAnswer(btn, isCorrect) {
      const parent = btn.parentElement;
      const buttons = parent.querySelectorAll('button');
      buttons.forEach(b => {
        b.disabled = true;
        b.classList.remove('hover:bg-blue-50');
      });

      if (isCorrect) {
        btn.classList.add('bg-emerald-100', 'text-emerald-800', 'border-emerald-300');
      } else {
        btn.classList.add('bg-rose-100', 'text-rose-800', 'border-rose-300');
      }
    }

    // Khởi chạy
    renderVideoList();
    renderVocabList();
    updateCompleteButtonState();
